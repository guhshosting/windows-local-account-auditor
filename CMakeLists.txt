cmake_minimum_required(VERSION 3.20)
project(windows_local_account_auditor VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32)
 message(FATAL_ERROR "Windows only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
option(ENABLE_ANALYZE "MSVC static analysis" OFF)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(MSVC)
 add_compile_options(/W4 /WX /permissive- /utf-8 /sdl /guard:cf)
 add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT)
 if(ENABLE_ANALYZE)
  add_compile_options(/analyze)
 endif()
endif()
add_library(audit src/core.cpp)
target_include_directories(audit PUBLIC src)
add_executable(windows-local-account-auditor src/main.cpp src/windows.cpp src/version.rc)
target_link_libraries(windows-local-account-auditor PRIVATE audit netapi32 advapi32)
add_executable(audit_tests tests/tests.cpp)
target_link_libraries(audit_tests PRIVATE audit advapi32)
enable_testing()
add_test(NAME fixtures COMMAND audit_tests)
add_test(NAME help COMMAND windows-local-account-auditor --help)
add_test(NAME version COMMAND windows-local-account-auditor --version)
add_test(NAME invalid_argument COMMAND windows-local-account-auditor --invalid)
set_tests_properties(invalid_argument PROPERTIES WILL_FAIL TRUE)
